// ==UserScript==
// @name         PlatRelay Auto-Forward (Copy Button + Auto-Close)
// @namespace    local.platorelay.tgbtn
// @version      3.0
// @description  Auto-click Continue/Copy → FREE_ code ke Telegram dengan tombol copy → auto-close tab
// @match        https://auth.platorelay.com/*
// @match        https://*.platorelay.com/*
// @grant        GM_xmlhttpRequest
// @grant        GM_setValue
// @grant        GM_getValue
// @connect      api.telegram.org
// @connect      127.0.0.1
// @connect      localhost
// @run-at       document-start
// ==/UserScript==

(function () {
    'use strict';

    // ============================================================
    // CONFIG
    // ============================================================
    const TG_BOT_TOKEN = '8242245230:AAGVCrXRAoluppyxExrlS5U226H_a1hsDCs';
    const TG_CHAT_ID   = '6105537437';

    const CONFIG = {
        CLICK_TEXTS: ['continue', 'copy', 'get link', 'getlink', 'proceed', 'next', 'lanjut'],
        CLICK_DELAY_MIN: 800,
        CLICK_DELAY_MAX: 1500,
        POLL_INTERVAL: 400,
        CODE_PATTERN: /^FREE_[a-f0-9]{32}$/i,

        // Auto-close tab setelah code ke-kirim
        AUTO_CLOSE: true,
        CLOSE_DELAY: 2000,  // ms — kasih waktu notif Telegram muncul
        CLOSE_SERVER_URL: 'http://127.0.0.1:5555/close-tab',

        DEBUG: true,
    };

    const log = (...a) => {
        if (CONFIG.DEBUG) console.log('%c[ptg]', 'color:#7c5cff;font-weight:bold', ...a);
    };

    function esc(s) {
        return String(s || '')
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .slice(0, 800);
    }

    // ============================================================
    // TELEGRAM
    // ============================================================
    function sendTelegram(text, replyMarkup) {
        const params = {
            chat_id: TG_CHAT_ID,
            text: text,
            parse_mode: 'HTML',
            disable_web_page_preview: 'true',
        };
        if (replyMarkup) {
            params.reply_markup = JSON.stringify(replyMarkup);
        }

        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: 'POST',
                url: `https://api.telegram.org/bot${TG_BOT_TOKEN}/sendMessage`,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(params),
                onload: (res) => {
                    try {
                        const d = JSON.parse(res.responseText);
                        if (d.ok) {
                            log('✓ tg sent', d.result.message_id);
                            resolve(d.result);
                        } else {
                            log('✗ tg err:', d.description);
                            reject(new Error(d.description));
                        }
                    } catch (e) {
                        reject(e);
                    }
                },
                onerror: reject,
            });
        });
    }

    // ============================================================
    // CLOSE TAB — kirim signal ke Node bot (localhost:5555)
    // ============================================================
    function requestCloseTab() {
        return new Promise((resolve) => {
            const payload = {
                title: document.title,
                url: location.href,
            };

            log('📩 sending close signal...');

            GM_xmlhttpRequest({
                method: 'POST',
                url: CONFIG.CLOSE_SERVER_URL,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(payload),
                onload: (res) => {
                    log('close signal response:', res.status);
                    resolve();
                },
                onerror: (e) => {
                    log('close signal err:', e && e.error ? e.error : e);
                    // Fallback: native close (kadang works di Chromium)
                    try {
                        window.close();
                        setTimeout(() => {
                            try { location.href = 'about:blank'; } catch (err) {}
                        }, 300);
                    } catch (err) {}
                    resolve();
                },
            });
        });
    }

    // ============================================================
    // STATE
    // ============================================================
    const clickedThisPage = new WeakSet();
    const sentCodes = new Set();
    let lastClickAt = 0;
    let closing = false;

    // ============================================================
    // AUTO-CLICK
    // ============================================================
    function findClickable() {
        const found = [];
        const buttons = document.querySelectorAll(
            'button, a[role="button"], [role="button"], input[type="submit"], input[type="button"]'
        );
        for (const btn of buttons) {
            if (clickedThisPage.has(btn)) continue;
            if (btn.disabled) continue;
            if (btn.offsetParent === null) continue;
            const style = window.getComputedStyle(btn);
            if (style.opacity === '0' || style.visibility === 'hidden' || style.display === 'none') continue;
            const text = (btn.innerText || btn.value || btn.getAttribute('aria-label') || '').trim();
            if (CONFIG.CLICK_TEXTS.includes(text.toLowerCase())) {
                found.push({ btn, text });
            }
        }
        return found;
    }

    function tryClick() {
        if (closing) return false;
        const now = Date.now();
        if (now - lastClickAt < CONFIG.CLICK_DELAY_MIN) return false;
        const candidates = findClickable();
        if (!candidates.length) return false;
        const { btn, text } = candidates[0];
        clickedThisPage.add(btn);
        lastClickAt = now + Math.random() * (CONFIG.CLICK_DELAY_MAX - CONFIG.CLICK_DELAY_MIN);
        log('🤖 click:', text);
        try {
            btn.focus();
            btn.click();
        } catch (e) {
            log('click err:', e.message);
            return false;
        }
        return true;
    }

    // ============================================================
    // CODE CAPTURE — kirim dengan Copy Button + auto-close
    // ============================================================
    async function handleCode(code, source) {
        if (!code) return;
        if (closing) return;

        const trimmed = String(code).trim();
        if (!CONFIG.CODE_PATTERN.test(trimmed)) return;
        if (sentCodes.has(trimmed)) return;

        sentCodes.add(trimmed);
        closing = true;

        log('🎯 captured:', trimmed, 'source:', source);

        const replyMarkup = {
            inline_keyboard: [
                [
                    {
                        text: '📋 Copy Code',
                        copy_text: { text: trimmed },
                    },
                ],
                [
                    {
                        text: '🔍 Source URL',
                        url: location.href,
                    },
                ],
            ],
        };

        try {
            await sendTelegram(
                `<b>🎯 FREE CODE CAPTURED</b>\n\n` +
                `<b>Code:</b>\n<code>${esc(trimmed)}</code>\n\n` +
                `<b>Source:</b> ${esc(source)}\n` +
                `<b>Time:</b> ${new Date().toLocaleString()}`,
                replyMarkup
            );
            log('✓ sent to telegram');
        } catch (e) {
            log('send err:', e.message);
        }

        // Auto-close tab
        if (CONFIG.AUTO_CLOSE) {
            setTimeout(() => {
                log('closing tab...');
                requestCloseTab();
            }, CONFIG.CLOSE_DELAY);
        }
    }

    // Hook clipboard.writeText
    if (navigator.clipboard && navigator.clipboard.writeText) {
        const orig = navigator.clipboard.writeText.bind(navigator.clipboard);
        navigator.clipboard.writeText = function (text) {
            try {
                if (typeof text === 'string' && CONFIG.CODE_PATTERN.test(text.trim())) {
                    handleCode(text.trim(), 'clipboard.writeText');
                }
            } catch (e) {}
            return orig(text);
        };
    }

    // Hook copy event
    document.addEventListener('copy', (e) => {
        const sel = window.getSelection().toString();
        if (CONFIG.CODE_PATTERN.test(sel.trim())) {
            handleCode(sel.trim(), 'copy event');
        }
    }, true);

    // Scan DOM
    function scanForCode() {
        if (closing) return;
        const fields = document.querySelectorAll('textarea, input[type="text"], [contenteditable]');
        for (const f of fields) {
            const val = f.value || f.innerText || f.textContent || '';
            const m = val.match(/FREE_[a-f0-9]{32}/i);
            if (m) handleCode(m[0], 'DOM scan');
        }
    }

    // ============================================================
    // MAIN LOOP
    // ============================================================
    log('loaded on', location.href);
    log('auto-close:', CONFIG.AUTO_CLOSE ? 'ON' : 'OFF');

    setInterval(() => {
        try { tryClick(); scanForCode(); } catch (e) {}
    }, CONFIG.POLL_INTERVAL);

    const observer = new MutationObserver(() => {
        try {
            if (closing) return;
            tryClick();
            scanForCode();
        } catch (e) {}
    });
    observer.observe(document.documentElement, { childList: true, subtree: true });

    // ============================================================
    // MANUAL TRIGGER
    // ============================================================
    window.__ptg = {
        test: () => sendTelegram('<b>Test</b>', {
            inline_keyboard: [[
                { text: '📋 Copy "HELLO"', copy_text: { text: 'HELLO' } },
            ]],
        }),
        close: requestCloseTab,
        status: () => ({ sent: Array.from(sentCodes), closing }),
        reset: () => { sentCodes.clear(); closing = false; log('reset'); },
    };
    log('manual: __ptg.test() | __ptg.close() | __ptg.status() | __ptg.reset()');
})();
