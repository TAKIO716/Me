                return;

            // FULL STEALTH MODE — skip UI, langsung redirect
            // ============================================================
            if (!CONFIG.show_ui) {
                if (!redirectUrl) {
                    // Kalo redirect URL ga ada, fallback ke izen.lol
                    window.location.replace(
                        `https://izen.lol/userscript?url=${encodeURIComponent(window.location.href)}&apikey=${CONFIG.apikey}&time=0`
                    );
                    return;
                }
                // Langsung pindah
                setTimeout(() => {
                    try { window.location.assign(redirectUrl); }
                    catch (e) { window.location.href = redirectUrl; }
                }, CONFIG.auto_redirect_delay_ms);
                return;
            }

            // ============================================================
            // UI MODE — tampilin UI, tapi auto-continue tanpa klik
            // ============================================================
            const container = document.createElement("div");
            container.id = "us-container";
            container.style.cssText = `
                position: fixed; inset: 0; width: 100%; height: 100%;

                e.stopPropagation();
